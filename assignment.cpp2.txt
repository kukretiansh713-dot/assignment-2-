These examples use **C++ character arrays** and functions from `<cstring>`. Remember to leave room in destination arrays for the terminating `'\0'`.

`strrev()` is not part of standard C++, so answers 10 and 11 use a small manual reversal loop instead.

## 1. Find string length using `strlen()`

```cpp
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char str[200];

    cout << "Enter a string: ";
    cin.getline(str, 200);

    cout << "Length = " << strlen(str);
    return 0;
}
```

## 2. Count vowels using `strlen()`

```cpp
#include <iostream>
#include <cstring>
#include <cctype>
using namespace std;

int main() {
    char str[200];
    int vowels = 0;

    cout << "Enter a string: ";
    cin.getline(str, 200);

    for (int i = 0; i < strlen(str); i++) {
        char ch = tolower(static_cast<unsigned char>(str[i]));
        if (ch == 'a' || ch == 'e' || ch == 'i' ||
            ch == 'o' || ch == 'u')
            vowels++;
    }

    cout << "Number of vowels = " << vowels;
    return 0;
}
```

## 4. Check whether two strings are equal using `strcpy()` and comparison

This copies the first string to another array, then compares the arrays with `strcmp()`.

```cpp
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char first[200], copy[200], second[200];

    cout << "Enter first string: ";
    cin.getline(first, 200);

    cout << "Enter second string: ";
    cin.getline(second, 200);

    strcpy(copy, first);

    if (strcmp(copy, second) == 0)
        cout << "Strings are equal.";
    else
        cout << "Strings are not equal.";

    return 0;
}
```

## 9. Arrange an array of strings alphabetically using `strcmp()`

```cpp
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    int n;
    char names[50][100], temp[100];

    cout << "Enter number of names: ";
    cin >> n;
    cin.ignore(10000, '\n');

    cout << "Enter " << n << " names:\n";
    for (int i = 0; i < n; i++)
        cin.getline(names[i], 100);

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (strcmp(names[j], names[j + 1]) > 0) {
                strcpy(temp, names[j]);
                strcpy(names[j], names[j + 1]);
                strcpy(names[j + 1], temp);
            }
        }
    }

    cout << "Names in alphabetical order:\n";
    for (int i = 0; i < n; i++)
        cout << names[i] << "\n";

    return 0;
}
```

## 10. Reverse a string

`strrev()` is available in some compilers, but is not standard C++. This version reverses the characters manually.

```cpp
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char str[200];

    cout << "Enter a string: ";
    cin.getline(str, 200);

    int length = strlen(str);

    for (int i = 0; i < length / 2; i++) {
        char temp = str[i];
        str[i] = str[length - 1 - i];
        str[length - 1 - i] = temp;
    }

    cout << "Reversed string: " << str;
    return 0;
}
```

## 11. Check whether a string is a palindrome

This makes a reversed copy and compares it with the original. The comparison is case-sensitive and includes spaces.

```cpp
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char str[200], reversed[200];

    cout << "Enter a string: ";
    cin.getline(str, 200);

    strcpy(reversed, str);

    int length = strlen(reversed);
    for (int i = 0; i < length / 2; i++) {
        char temp = reversed[i];
        reversed[i] = reversed[length - 1 - i];
        reversed[length - 1 - i] = temp;
    }

    if (strcmp(str, reversed) == 0)
        cout << "The string is a palindrome.";
    else
        cout << "The string is not a palindrome.";

    return 0;
}
```

## 12. Find the first occurrence of a character using `strchr()`

```cpp
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char str[200], ch;

    cout << "Enter a string: ";
    cin.getline(str, 200);

    cout << "Enter the character to find: ";
    cin >> ch;

    char *result = strchr(str, ch);

    if (result != nullptr)
        cout << "First occurrence is at position " << (result - str) + 1;
    else
        cout << "Character not found.";

    return 0;
}
```

## 16. Count words using `strlen()` and spaces

This treats any run of spaces as a separator and does not count leading or trailing spaces as words.

```cpp
#include <iostream>
#include <cstring>
using namespace std;

int main() {
    char str[300];
    int words = 0;
    bool insideWord = false;

    cout << "Enter a string: ";
    cin.getline(str, 300);

    for (int i = 0; i < strlen(str); i++) {
        if (str[i] != ' ' && !insideWord) {
            words++;
            insideWord = true;
        } else if (str[i] == ' ') {
            insideWord = false;
        }
    }

    cout << "Number of words = " << words;
    return 0;
}
```

## 17. Remove all vowels from a string

```cpp
#include <iostream>
#include <cstring>
#include <cctype>
using namespace std;

int main() {
    char str[300];
    int writeIndex = 0;

    cout << "Enter a string: ";
    cin.getline(str, 300);

    for (int readIndex = 0; readIndex < strlen(str); readIndex++) {
        char ch = tolower(static_cast<unsigned char>(str[readIndex]));

        if (ch != 'a' && ch != 'e' && ch != 'i' &&
            ch != 'o' && ch != 'u') {
            str[writeIndex] = str[readIndex];
            writeIndex++;
        }
    }

    str[writeIndex] = '\0';

    cout << "String without vowels: " << str;
    return 0;
}
```

## 20. Check whether two strings are anagrams

This version ignores spaces and letter case. It uses character counting after removing spaces.

```cpp
#include <iostream>
#include <cstring>
#include <cctype>
using namespace std;

int main() {
    char first[300], second[300];
    int counts[256] = {};

    cout << "Enter first string: ";
    cin.getline(first, 300);

    cout << "Enter second string: ";
    cin.getline(second, 300);

    for (int i = 0; i < strlen(first); i++) {
        unsigned char ch = static_cast<unsigned char>(first[i]);
        if (!isspace(ch))
            counts[tolower(ch)]++;
    }

    for (int i = 0; i < strlen(second); i++) {
        unsigned char ch = static_cast<unsigned char>(second[i]);
        if (!isspace(ch))
            counts[tolower(ch)]--;
    }

    bool anagrams = true;
    for (int i = 0; i < 256; i++) {
        if (counts[i] != 0) {
            anagrams = false;
            break;
        }
    }

    if (anagrams)
        cout << "The strings are anagrams.";
    else
        cout << "The strings are not anagrams.";

    return 0;
}
```
